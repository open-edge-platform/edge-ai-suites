Terminal 1:
  make latency-test-relay                 Start relay node

Terminal 2:
  make latency-test-main                  Run latency test (1000 Hz, 60s)
  make latency-quick                      Quick 30s test
  make latency-test-main RATE=2000        Custom rate

With parameters:
  make latency-test-main RATE=2000 DURATION=60 RELIABILITY=RELIABLE

make grafana-status                       Check status

Export a session to Grafana:
  make grafana-export SESSION=20260306_154140
  make grafana-export                      # uses latest session
  make grafana-export-live                 # live mode
